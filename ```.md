WITH raw_nicknames AS (

SELECT explode(split(:model_nicknames, ',')) AS nickname_raw

),

nicknames AS (

SELECT lower(trim(nickname_raw)) AS nickname

FROM raw_nicknames

WHERE trim(nickname_raw) <> ''

),

selected_models AS (

SELECT model_id, nickname

FROM (

SELECT

mts.id AS model_id,

mts.nickname,

row_number() OVER (

PARTITION BY lower(mts.nickname)

ORDER BY

CASE WHEN mts.archived_at IS NULL THEN 0 ELSE 1 END,

mts.ingested_at DESC,

mts.updated_at DESC

) AS rn

FROM hive_metastore.raw__model_catalogue_sync.model_training_sessions mts

JOIN nicknames n

ON lower(mts.nickname) = n.nickname

) t

WHERE rn = 1

),

latest_checkpoint AS (

SELECT

r.model_id,

MAX(try_cast(r.model_checkpoint_num AS BIGINT)) AS latest_checkpoint_num

FROM hive_metastore.raw__gen2.robot_start_av_run r

JOIN selected_models sm

ON sm.model_id = r.model_id

GROUP BY r.model_id

),

model_runs AS (

SELECT DISTINCT

r.run_id,

r.model_id,

CAST(r.model_checkpoint_num AS STRING) AS model_checkpoint_num

FROM hive_metastore.raw__gen2.robot_start_av_run r

JOIN latest_checkpoint lc

ON r.model_id = lc.model_id

AND try_cast(r.model_checkpoint_num AS BIGINT) = lc.latest_checkpoint_num

),

meta AS (

SELECT

mr.run_id,

mr.model_id,

sm.nickname AS model_nickname,

mr.model_checkpoint_num,

rm.start_time AS start_time_struct,

rm.end_time AS end_time_struct,

CASE WHEN mr.run_id LIKE '%gen2-av%' THEN 'AV' ELSE 'DC' END AS run_type,

(rm.end_time.seconds - rm.start_time.seconds) / 60.0 AS run_duration_min,

CAST(((rm.start_time.seconds) * 1000000000L + rm.start_time.nanos) / 1000000 AS BIGINT) AS start_time_ms,

CAST(((rm.end_time.seconds) * 1000000000L + rm.end_time.nanos) / 1000000 AS BIGINT) AS end_time_ms,

COALESCE(rm.vehicle_fleet360_uuid, 'please_set_host') AS host

FROM model_runs mr

JOIN selected_models sm

ON sm.model_id = mr.model_id

JOIN hive_metastore.raw__gen2.robot_metadata rm

ON rm.run_id = mr.run_id

)

  

SELECT

model_nickname,

model_id,

model_checkpoint_num,

concat('https://console.sso.wayve.ai/model/', model_id, '@', model_checkpoint_num) AS model_link,

run_id,

run_type,

run_duration_min,

start_time_ms,

end_time_ms,

from_unixtime(start_time_ms / 1000.0) AS start_time,

from_unixtime(end_time_ms / 1000.0) AS end_time,

host,

  

concat(

'https://app.datadoghq.eu/dashboard/h25-6jt-saq/parking-dashboard?live=false&tpl_var_host%5B0%5D=',

host, '&from_ts=', start_time_ms, '&to_ts=', end_time_ms

) AS datadog_parking_dashboard_link,

  

concat('https://console.sso.wayve.ai/run/', run_id) AS console_link,

  

concat(

'https://app.foxglove.dev/wayve-on-prem/p/prj_0dX18KZdVHg1fmmI/view?ds=foxglove-stream',

'&ds.deviceName=', element_at(split(run_id, '/'), 1),

'&ds.end=', url_encode(regexp_replace(concat(

date_format(to_timestamp(end_time_struct.seconds + end_time_struct.nanos / 1000000000.0), 'yyyy-MM-dd'),

'T',

date_format(to_timestamp(end_time_struct.seconds + end_time_struct.nanos / 1000000000.0), 'HH:mm:ss.SSS'),

'Z'

), '\\s+', '')),

'&ds.start=', url_encode(regexp_replace(concat(

date_format(to_timestamp(start_time_struct.seconds + start_time_struct.nanos / 1000000000.0), 'yyyy-MM-dd'),

'T',

date_format(to_timestamp(start_time_struct.seconds + start_time_struct.nanos / 1000000000.0), 'HH:mm:ss.SSS'),

'Z'

), '\\s+', '')),

'&layoutId=lay_0e7VD4WIKDQGU73Y',

'&time=', url_encode(regexp_replace(concat(

date_format(

to_timestamp(

((start_time_struct.seconds + start_time_struct.nanos / 1000000000.0) +

(end_time_struct.seconds + end_time_struct.nanos / 1000000000.0)) / 2.0

),

'yyyy-MM-dd'

),

'T',

date_format(

to_timestamp(

((start_time_struct.seconds + start_time_struct.nanos / 1000000000.0) +

(end_time_struct.seconds + end_time_struct.nanos / 1000000000.0)) / 2.0

),

'HH:mm:ss.SSS'

),

'Z'

), '\\s+', ''))

) AS foxglove_url,

  

concat(

'https://adb-7835963732836817.17.azuredatabricks.net/dashboardsv3/01efa03c10f61a84ac20842643377b9c/published?o=7835963732836817&f_6850309d%7Ee013df3c=',

url_encode(url_encode(trim(run_id)))

) AS gen2_run_inspection_dashboard_link

  

FROM meta

ORDER BY start_time_ms DESC;
